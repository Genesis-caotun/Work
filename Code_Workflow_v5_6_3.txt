function doGet(e){
  const p=(e&&e.parameter)||{}, cb=String(p.callback||'callback');
  if(!/^[a-zA-Z_$][\w$\.]*$/.test(cb))return ContentService.createTextOutput('invalid callback');
  let r;try{
    if(p.action==='login')r=login_(p.idToken);
    else if(p.action==='register')r=register_(p.idToken,p);
    else if(p.action==='logout')r=logout_(p.authToken);
    else if(p.action==='dashboard')r=dashboard_(p.authToken);
    else if(p.action==='uploadStatus')r=uploadStatus_(p.authToken,p.requestId);
    else if(p.action==='getPreviousFile')r=getPreviousFile_(p.authToken,p.caseId,p.fileType);
    else if(p.action==='saveUser')r=saveUser_(p.authToken,p);
    else if(p.action==='deleteUser')r=deleteUser_(p.authToken,p.email);
    else if(p.action==='createCase')r=createCase_(p.authToken,p);
    else if(p.action==='markRead')r=markRead_(p.authToken,p.caseId);
    else if(p.action==='editCase')r=editCase_(p.authToken,p);
    else if(p.action==='deleteCase')r=deleteCase_(p.authToken,p.caseId);
    else r={ok:true,message:'Work Approval API',version:APP_VERSION};
  }catch(x){r={ok:false,error:x.message||String(x)}}
  return ContentService.createTextOutput(cb+'('+JSON.stringify(r)+')').setMimeType(ContentService.MimeType.JAVASCRIPT);
}

function doPost(e){
  const p=(e&&e.parameter)||{},requestId=String(p.requestId||'');
  let effectiveUser='',activeUser='';try{effectiveUser=Session.getEffectiveUser().getEmail()}catch(x){}try{activeUser=Session.getActiveUser().getEmail()}catch(x){}
  console.log('UPLOAD_REQUEST ['+requestId+'] effectiveUser='+effectiveUser+' activeUser='+activeUser+' case='+String(p.caseId||'')+' type='+String(p.fileType||'')+' file='+String(p.fileName||'')+' contentLength='+String((e&&e.contentLength)||0)+' base64Chars='+String((p.fileData||'').length));
  let r;try{
    if(p.action==='upload')r=upload_(p.authToken,p);
    else if(p.action==='returnCase')r=returnCase_(p.authToken,p);
    else throw new Error('不支援的操作');
  }catch(x){
    r={ok:false,error:x.message||String(x)};
    console.error('UPLOAD_ERROR ['+requestId+'] '+r.error);
  }
  if(/^[a-zA-Z0-9_-]{10,120}$/.test(requestId)){
    try{CacheService.getScriptCache().put('UPLOAD_'+requestId,JSON.stringify({tokenHash:hash_(p.authToken||''),result:r}),600)}
    catch(x){console.log('upload status: '+x.message)}
  }
  const json=JSON.stringify(Object.assign({},r,{requestId:requestId})).replace(/</g,'\\u003c');
  return HtmlService.createHtmlOutput('<script>parent.postMessage('+json+',"'+APP_ORIGIN+'");</script>')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function uploadStatus_(token,requestId){
  requireUser_(token);
  const id=String(requestId||'');
  if(!/^[a-zA-Z0-9_-]{10,120}$/.test(id))throw new Error('上傳查詢碼不正確');
  const raw=CacheService.getScriptCache().get('UPLOAD_'+id);
  if(!raw)return{ok:true,done:false};
  const saved=JSON.parse(raw);
  if(saved.tokenHash!==hash_(token))throw new Error('無權查詢這次上傳');
  return{ok:true,done:true,result:saved.result||{ok:false,error:'找不到上傳結果'}};
}
